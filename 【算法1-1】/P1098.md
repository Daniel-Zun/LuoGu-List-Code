#include<iostream>

using namespace std;

int p1, p2, p3;

void Expand(char x, char y, int trans)
{
    int num = y - x;
    if (num > 1) 
    {
        string ans;
        for (int i = 1; i < num; i++)
            for (int j = 1; j <= p2; j++)
                if (p1 == 3) ans.push_back('*');
                else
                {
                    if (trans == 2 && p1 == 2) ans.push_back(char(x - ('a' - 'A') + i));
                    else ans.push_back(char(x + i));
                }
        if (p3 == 1) cout << ans;
        else 
            for (int k = (int)ans.size() - 1; k >= 0; k--) cout << ans[k];
    }
}

int main()
{
    cin >> p1 >> p2 >> p3;
    string s;
    cin >> s;
    int i = 0;
    while(i + 1 < s.size())
    {
        cout << s[i];
        if (s[i + 1] == '-')
        {
            int trans = 0, num = s[i + 2] - s[i];
            if (s[i] >= '0' && s[i + 2] <= '9' && num > 0) trans = 1;
            else if (s[i] >= 'a' && s[i + 2] <= 'z' && num > 0) trans = 2;
            if (trans > 0) 
            {
                Expand(s[i], s[i + 2], trans);
                i++;
            }
        }
        i++;
    }
    cout << s[i];
    return 0;
}
