```cpp
#include<iostream>
#include<algorithm>

using namespace std;

string s[7] = {"BangZhu", "FuBangZhu", "HuFa", "ZhangLao", "TangZhu", "JingYing", "BangZhong"};

struct Party
{
    int work, height, num;
    bool final;
    string name, statue;
    void Input(int n);
    void Output() 
    {
        cout << name << " " << statue << " " << height << endl;;
    }
};

void Party::Input(int n)
{
    cin >> name >> statue >> work >> height;
    num = n;
}

int Zw(string o)
{
    for (int i = 0; i < 7; i++)
        if (s[i] == o) return i;
}

bool cmp(Party a, Party b)
{
    if (a.work == b.work) return a.num < b.num;
    else return a.work > b.work;
}

bool cmp_final(Party a, Party b)
{
    if (Zw(a.statue) != Zw(b.statue)) return Zw(a.statue) < Zw(b.statue);
    else if (a.height != b.height) return a.height > b.height;
    else return a.num < b.num;
}

int main()
{
    int n;
    cin >> n;
    Party man[120];
    for (int i = 1; i <= n; i++)
        man[i].Input(i);
    sort(man + 1, man + n + 1, cmp);
    int q = 1;
    for (int i = 1; i <= n; i++)
    {
        if (man[i].statue == "BangZhu" || man[i].statue == "FuBangZhu") continue;
        else if (q < 3) man[i].statue = s[2];
        else if (q > 2 && q < 7) man[i].statue = s[3];
        else if (q > 6 && q < 14) man[i].statue = s[4];
        else if (q > 13 && q < 39) man[i].statue = s[5];
        else man[i].statue = s[6];
        q++;
    }
    sort(man + 1, man + n + 1, cmp_final);
    for (int i = 1; i <= n; i++)
        man[i].Output();
    return 0;
}
```
