#include<iostream>
#include<vector>
#include<cstring>

using namespace std;

int main()
{
    int n;
    cin >> n;
    //特殊判断
    if (n < 5) 
    {
        cout << n;
        return 0;
    }
    //贪心拆分
    int r = 2, sum = 0;
    vector<int> num;
    while(n - sum >= r)
    {
        num.push_back(r);
        sum += r;
        r++;
    }
    int rest = n - sum, pos = num.size() - 1;
    if (rest == num[pos]) 
    {
        num[pos]++;
        rest--;
    }
    while(rest--)
    {
        num[pos]++;
        pos--;
    }
    //输出拆分结果
    for (int i : num)
        cout << i << " ";
    cout << endl;
    //高精度乘法
    int number[10000];
    memset(number, 0, sizeof(number));
    number[0] = 1;
    //竖式乘法
    for (int i = 0; i < num.size(); i++)
    {
        for (int j = 0; j < 10000; j++)
            number[j] *= num[i];
        for (int j = 0; j < 10000; j++)
            if (number[j] > 9)
            {
                number[j + 1] += number[j] / 10;
                number[j] = number[j] % 10;
            }
    }
    //输出
    int ans = 0, final = 9999;
    while(number[final] == 0 && final > 0) final--;
    for (int i = final; i >= 0; i--)
        cout << number[i];
    return 0;
}
