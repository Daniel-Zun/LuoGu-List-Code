**代码：**
```cpp
#include<iostream>
#include<vector>
#include<cstring>

using namespace std;

int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        int num[10000];
        memset(num, 0, sizeof(num));
        int n, target;
        cin >> n >> target;
        num[0] = 1;
        //竖式乘法
        for (int i = 1; i <= n; i++)
        {
            for (int j = 0; j < 10000; j++)
                num[j] *= i;
            for (int j = 0; j < 10000; j++)
                if (num[j] > 9)
                {
                    num[j + 1] += num[j] / 10;
                    num[j] = num[j] % 10;
                }
        }
        //输出
        int ans = 0, final = 9999;
        while(num[final] == 0 && final > 0) final--;
        for (int i = final; i >= 0; i--)
            if (num[i] == target) ans++;
        cout << ans << endl;
    }
    return 0;
}
```
